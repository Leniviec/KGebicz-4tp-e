uczniowie = []

plik = open("uczniowie.txt", "r")

for linia in plik:
    uczniowie.append(linia.strip())

plik.close()

while True:
    print("\n1. wyswietl liste uczniow")
    print("2. posortuj alfabetycznie")
    print("3. usun ucznia")
    print("4. zakoncz program")

    wybor = input("wybierz opcje: ")

    if wybor == "1":
        for uczen in uczniowie:
            print(uczen)

    elif wybor == "2":
        uczniowie.sort()

        plik = open("uczniowie.txt", "w")
        for uczen in uczniowie:
            plik.write(uczen + "\n")
        plik.close()

        print("lista posortowana")

    elif wybor == "3":
        imie = input("podaj imie i nazwisko ucznia: ")

        if imie in uczniowie:
            uczniowie.remove(imie)

            plik = open("uczniowie.txt", "w")
            for uczen in uczniowie:
                plik.write(uczen + "\n")
            plik.close()

            print("uczen usuniety")
        else:
            print("nie ma takiego ucznia")

    elif wybor == "4":
        print("koniec programu")
        break

    else:
        print("nie ma takiej opcji")

input("nacisnij enter, aby zamknac")
