package com.example.rejestr;

import android.os.Bundle;
import android.util.Patterns;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import java.io.FileOutputStream;
import java.io.IOException;
import java.nio.charset.StandardCharsets;

public class MainActivity extends AppCompatActivity {

    private EditText Imie;
    private EditText Nazwisko;
    private EditText Email;
    private EditText Haslo;
    private Button btnZarejestruj;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        Imie = findViewById(R.id.Imie);
        Nazwisko = findViewById(R.id.Nazwisko);
        Email = findViewById(R.id.Email);
        Haslo = findViewById(R.id.Haslo);
        btnZarejestruj = findViewById(R.id.btnZarejestruj);

        btnZarejestruj.setOnClickListener(v -> zarejestruj());
    }

    private void zarejestruj() {

        String imie = Imie.getText().toString().trim();
        String nazwisko = Nazwisko.getText().toString().trim();
        String email = Email.getText().toString().trim();
        String haslo = Haslo.getText().toString();

        if(imie.isEmpty() || nazwisko.isEmpty() || email.isEmpty() || haslo.isEmpty()){
            System.out.println("Uzupełnij wszystkie pola!!!");
            System.exit(0);
        };
        boolean zawiera;
        if (email.contains("@") && email.contains(".")) {
            zawiera = true;
        } else {
            System.out.println("Podaj poprawny adres email!!");
            System.exit(0);
        };

        if (haslo.length() < 8
                || !haslo.matches(".*[A-Z].*")
                || !haslo.matches(".*[a-z].*")
                || !haslo.matches(".*[!@#$%^&*()_+{}?;':<>,./\\-=].*")) {

            System.out.println("Hasło musi mieć co najmniej 8 znaków, dużą i małą literę oraz znak specjalny!!!");
            System.exit(0);
        }
        System.out.println("Dane są prawidłowe");
        System.exit(0);
    }
    
    private void Zapisz(){
        String imie = Imie.getText().toString().trim();
        String nazwisko = Nazwisko.getText().toString().trim();
        String email = Email.getText().toString().trim();
        String haslo = Haslo.getText().toString().trim();
        
        String data = "Imię: " + imie + " Nazwisko: " + nazwisko + " Email:  " + email + " Hasło: " + haslo;
        
        try(FileOutputStream fos = openFileOutput(
                "dane.txt",
                MODE_APPEND
        )) {
            fos.write(data.getBytes(StandardCharsets.UTF_8));
        }catch (IOException e){
            Toast.makeText(
                    this, "błąd zapisu",Toast.LENGTH_SHORT
            ).show();
        }
    };
};
